JAM DAN TANGGAL ===== function updateClock() { const now = new Date(); const hours = String(now.getHours()).padStart(2, '0'); const minutes = String(now.getMinutes()).padStart(2, '0'); document.getElementById('clock').textContent = `${hours}:${minutes}`; const days = ['Minggu', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu']; const months = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember']; document.getElementById('date').textContent = `${days[now.getDay()]}, ${now.getDate()} ${months[now.getMonth()]} ${now.getFullYear()}`; } updateClock(); setInterval(updateClock, 1000); // ===== 2. BUKA & KUNCI LAYAR ===== const lockscreen = document.getElementById('lockscreen'); const homeScreen = document.getElementById('home-screen'); function unlockScreen() { lockscreen.classList.add('unlocked'); } function lockScreen() { lockscreen.classList.remove('unlocked'); } // Fungsi deteksi geser (Mouse & Touchscreen) function addSwipeListener(element, direction, callback) { let startY = 0; let isPointerDown = false; element.addEventListener('touchstart', (e) => { startY = e.touches[0].clientY; }, { passive: true }); element.addEventListener('touchend', (e) => { const endY = e.changedTouches[0].clientY; const distance = startY - endY; if (direction === 'up' && distance > 60) callback(); if (direction === 'down' && distance < -60) callback(); }); element.addEventListener('mousedown', (e) => { startY = e.clientY; isPointerDown = true; }); element.addEventListener('mouseup', (e) => { if (!isPointerDown) return; isPointerDown = false; const endY = e.clientY; const distance = startY - endY; if (direction === 'up' && distance > 60) callback(); if (direction === 'down' && distance < -60) callback(); }); } addSwipeListener(lockscreen, 'up', unlockScreen); // Geser ke atas = Buka addSwipeListener(homeScreen, 'down', lockScreen); // Geser ke bawah = Kunci // ===== 3.
